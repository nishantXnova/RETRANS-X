# Chunk-size sweep (optional): which C_CHUNK is fastest at T=8192?
import torch, time

device = 'cuda'
B, T, H, N = 4, 8192, 128, 8
torch.manual_seed(0)
u = torch.randn(B, T, H, device=device).requires_grad_()
dt = (torch.rand(B, T, H, device=device).clamp_min(1e-3) * 0.1).requires_grad_()
A = (-torch.rand(H, N, device=device).clamp_min(1e-3)).requires_grad_()
Bp = torch.randn(B, T, N, device=device).requires_grad_()
C = torch.randn(B, T, N, device=device)
D = torch.randn(H, device=device)

def run(chunk):
    h = ts.ChunkedSSMScanFn.apply(u, dt, A, Bp, T, chunk)
    y = (h * C.unsqueeze(2)).sum(-1) + D * u
    y.pow(2).mean().backward()
    for t in (dt, A, Bp):
        t.grad = None

print(f'T={T} H={H} N={N} B={B}: fwd+bwd per C_CHUNK')
for chunk in (32, 64, 128, 256, 512, 1024):
    if T % chunk:
        continue
    for _ in range(3):
        run(chunk)
    torch.cuda.synchronize()
    t0 = time.perf_counter()
    for _ in range(10):
        run(chunk)
    torch.cuda.synchronize()
    ms = (time.perf_counter() - t0) / 10 * 1000
    print(f'  C_CHUNK={chunk:>4}: {ms:7.2f} ms')

# Honest end-to-end re-measurement for index.html §5.4 (replaces stale timings
# that were logged without torch.cuda.synchronize). Fresh model per row,
# fwd+bwd per iter, sync after each batch. All rows below have B*H >= 512, so
# auto and fused pick the same fused path here (chunked only engages at B*H<=128).
# Run inside the stream_colab.ipynb training environment (has md, ts, device).
def bench_e2e(B, T, mode, iters=5):
    m = md.Stream(md.StreamConfig(
        n_embd=128, n_layer=4, ssm_d_state=8, n_predict=4,
        block_size=T, dropout=0.0, bias=False,
    )).to(device)
    if mode == 'jit':
        pass  # default JIT scan
    elif mode == 'triton':
        ts.enable_triton(m, True, fused=True)
    elif mode == 'auto':
        ts.enable_triton(m, True, auto=True)
    m.train()
    x = torch.randint(0, 256, (B, T), device=device)
    y = torch.randint(0, 256, (B, T), device=device)
    for _ in range(2):  # warmup
        _, loss = m(x, targets=y); loss.backward(); m.zero_grad()
    torch.cuda.synchronize()
    t0 = time.perf_counter()
    for _ in range(iters):
        _, loss = m(x, targets=y); loss.backward(); m.zero_grad()
    torch.cuda.synchronize()
    ms = (time.perf_counter() - t0) / iters * 1000
    tok_s = B * T / (ms / 1000)
    return ms, tok_s

rows = [('jit', 4, 4096), ('triton', 4, 4096), ('triton', 4, 16384),
        ('auto', 4, 4096), ('auto', 4, 16384)]
print(f'{"mode":>7} {"B":>2} {"T":>6}  {"ms/iter":>8}  {"tok/s":>9}')
results = {}
for mode, B, T in rows:
    ms, tok_s = bench_e2e(B, T, mode)
    results[(mode, B, T)] = (ms, tok_s)
    print(f'{mode:>7} {B:>2} {T:>6}  {ms:8.1f}  {tok_s:9.0f}')

jit_ms = results[('jit', 4, 4096)][0]
print('\nspeedup vs JIT (T=4096,B=4):')
for mode, B, T in rows:
    ms = results[(mode, B, T)][0]
    print(f'  {mode} T={T} B={B}: {jit_ms/ms:.2f}x')
print('\nPaste the ms/iter + tok/s rows into index.html §5.4 and drop the stale warning.')


# ===================== CELL 10 =====================
# HOTFIX (Colab doesn't get new commits): the shipped auto wrapper shadowed its
# own B argument (input-projection tensor) with 'B, T, H = u.shape', passing the
# batch-size int into the kernels -> "'int' object has no attribute 'stride'".
# Patched version below renames it to Bs. enable_triton reads the wrapper as a
# module global at call time, so reassigning ts.auto_triton_wrapped_scan is
# enough — no need to re-pull the repo. Run BEFORE any auto=True call.
def _auto_scan_fixed(self, u, dt, A, B, C):
    D = self.D.to(u.dtype)
    Bs, T, H = u.shape
    path = ts.auto_scan_path(Bs, T, H)
    if getattr(ts, 'AUTO_TRACE', False):
        ts.AUTO_PATH_LOG.append((Bs, T, H, path))
    if path == 'chunked':
        h = ts.ChunkedSSMScanFn.apply(u, dt, A, B, T, ts.CHUNK_DEFAULT)
    else:
        h = ts.FusedSSMScanFn.apply(u, dt, A, B, T)
    y = (h * C.unsqueeze(2)).sum(-1) + D * u
    return y, (h[:, -1].detach(), None)

ts.auto_triton_wrapped_scan = _auto_scan_fixed
print('patched ts.auto_triton_wrapped_scan (B-shadowing fix applied)')

# Verify the fixed wrapper in both dispatch regimes (B*H=128 chunked, B*H=256 fused).
class _Stub(torch.nn.Module):
    pass
stub = _Stub()
stub.D = torch.randn(64, device='cuda')
ok = True
for B in (2, 4):
    T, H, N = 1024, 64, 8
    label = 'chunked' if ts.auto_scan_path(B, T, H) == 'chunked' else 'fused'
    torch.manual_seed(0)
    u = torch.randn(B, T, H, device='cuda'); u.requires_grad_()
    dt = torch.rand(B, T, H, device='cuda').clamp_min(1e-3) * 0.1; dt.requires_grad_()
    A = -torch.rand(H, N, device='cuda').clamp_min(1e-3); A.requires_grad_()
    Bp = torch.randn(B, T, N, device='cuda'); Bp.requires_grad_()
    C = torch.randn(B, T, N, device='cuda')
    y_w, _ = _auto_scan_fixed(stub, u, dt, A, Bp, C)
    y_w.pow(2).mean().backward()
    gw = {n: t.grad.clone() for n, t in [('u', u), ('dt', dt), ('A', A), ('B', Bp)]}
    for t in (u, dt, A, Bp): t.grad = None
    h_f = ts.FusedSSMScanFn.apply(u, dt, A, Bp, T)
    y_f = (h_f * C.unsqueeze(2)).sum(-1) + stub.D * u
    y_f.pow(2).mean().backward()
    gf = {n: t.grad for n, t in [('u', u), ('dt', dt), ('A', A), ('B', Bp)]}
    worst = (y_w - y_f).abs().max().item()
    for n in ('u', 'dt', 'A', 'B'):
        worst = max(worst, (gw[n] - gf[n]).abs().max().item())
    ok &= worst < 1e-4
    print(f'  auto B*H={B*H} -> {label}: worst diff {worst:.2e} {"OK" if worst < 1e-4 else "FAIL"}')
print('WRAPPER CHECK:', 'PASS' if ok else 'FAIL')
print('Now re-run CELL 9 for the honest §5.4 timings (auto rows will work).')
